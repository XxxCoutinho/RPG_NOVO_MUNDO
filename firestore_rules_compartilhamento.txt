rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function userEmail() {
      return signedIn() && request.auth.token.email != null
        ? request.auth.token.email.lower()
        : '';
    }

    function isCampaignOwner(data) {
      return signedIn() && data.ownerUid == request.auth.uid;
    }

    function isCampaignMember(data) {
      return signedIn() && (
        data.ownerUid == request.auth.uid ||
        request.auth.uid in data.memberUids ||
        userEmail() in data.invitedEmails
      );
    }

    match /campaigns/{campaignId} {
      allow read: if isCampaignMember(resource.data);

      allow create: if signedIn()
        && request.resource.data.ownerUid == request.auth.uid
        && request.auth.uid in request.resource.data.memberUids;

      allow update: if isCampaignOwner(resource.data)
        || isCampaignMember(resource.data)
        || (
          signedIn()
          && request.resource.data.inviteToken == resource.data.inviteToken
          && request.auth.uid in request.resource.data.memberUids
        );

      allow delete: if isCampaignOwner(resource.data);
    }

    match /characters/{characterId} {
      allow read: if signedIn() && (
        resource.data.ownerUid == request.auth.uid ||
        resource.data.campaignOwnerUid == request.auth.uid ||
        request.auth.uid in resource.data.memberUids ||
        userEmail() in resource.data.invitedEmails
      );

      allow create: if signedIn() && (
        request.resource.data.ownerUid == request.auth.uid ||
        request.resource.data.campaignOwnerUid == request.auth.uid
      );

      allow update, delete: if signedIn() && (
        resource.data.ownerUid == request.auth.uid ||
        resource.data.campaignOwnerUid == request.auth.uid
      );
    }

    match /sheets/{sheetId} {
      allow read: if signedIn() && (
        resource.data.ownerUid == request.auth.uid ||
        request.auth.uid in resource.data.memberUids ||
        userEmail() in resource.data.invitedEmails
      );

      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;

      allow update, delete: if signedIn() && resource.data.ownerUid == request.auth.uid;
    }

    match /usuarios/{userId}/{document=**} {
      allow read, write: if signedIn() && userId == request.auth.uid;
    }
  }
}
